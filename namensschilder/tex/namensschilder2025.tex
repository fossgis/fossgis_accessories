% Sichtbare Seite des Badges
\newcommand{\badgesichtbar}[5]{\ticket{%
    % #1 = badgetype
    % #2 = tickettype
    % #3 = badgename
    % #4 = badgezusatz
    % #5 = AV
    %
    % Breite DIN A 7 74mm x 105mm
    % Hintergrundbild
    \put(0,0){\includegraphics[width=105mm]{img/fossgis2025_background.pdf}}

    % ticket type?
    \IfStrEqCase{#1}{%
        {CONF}{\ticketLogoCONF}%
        {HACK}{\ticketLogoHACK}%
        {OSM}{\ticketLogoOSM}%
        {SPRINT}{\tickedLogoSPRINT}%
    }[]% Default case

   % Helfende? zeichne rote line ein
   \IfStrEq{#2}{Konferenzticket für Helfende (bezahlt)}{%
        \put(30mm,57mm){
        \begin{tikzpicture}
                %\draw[orange, line width=2mm] (0,0) -- (5.2,0);
                \draw[orange, ultra thick] (0,0) -- (5.3,0);
            \end{tikzpicture}
        }
        %\put(82mm,65mm){
        %    \parbox{10mm}{
        %    \raggedleft
        %    \Huge \textcolor{orange}{H}
        %}}
   }{}
    % FOSSGIS Abendveranstaltung? dann zeige event.png oben rechts
    \IfStrEq{#1}{CONF}{%
        \put(85mm,65mm){
            \parbox{10mm}{
            \raggedleft
            \ifthenelse{\equal{#5}{True}}{\includegraphics[width=5mm]{img/logo_abendveranstaltung.pdf}}{}
            }
        }
    }{}
    % Zeige namen und zusatz (firma/nickname etc.) and
    \put(5mm,32mm){
       \adjustbox{%fbox, % zeichnet rahmen (für Positionierung)
            color=gray,
            minipage=[c][40mm]{90mm},
            % max totalheight=70mm, max width=95mm
            }{%
            \begin{spacing}{0.5}
            \begin{center}
            \Huge \textbf{#3} \\
            \vspace{1ex}
            \large{#4}
            \end{center}
            \end{spacing}
        }
      }

}}
